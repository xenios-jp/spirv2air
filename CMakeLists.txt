cmake_minimum_required(VERSION 3.20)
project(spirv2air VERSION 1.0.4 LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "This version requires the macOS SDK (CommonCrypto).")
endif()
set(SPIRV2AIR_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(SPIRV2AIR_TOP_LEVEL ON)
  if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
  endif()
  include(CTest)
endif()
include(CMakeDependentOption)
option(SPIRV2AIR_BUILD_TRANSLATOR "Build the SPIR-V translator (needs SPIRV-Headers); OFF builds only airbc, mtlb and airkit" ON)
cmake_dependent_option(SPIRV2AIR_BUILD_CLI "Build the command-line translator" ${SPIRV2AIR_TOP_LEVEL}
                       SPIRV2AIR_BUILD_TRANSLATOR OFF)
cmake_dependent_option(SPIRV2AIR_BUILD_TESTS "Build hand-written tests" ${SPIRV2AIR_TOP_LEVEL}
                       SPIRV2AIR_BUILD_TRANSLATOR OFF)
option(SPIRV2AIR_BUILD_GPU_TESTS "Build tests that require a Metal device" ON)
cmake_dependent_option(SPIRV2AIR_BUILD_AMBER_RUNNER "Build the Amber compute conformance runner (needs a Metal device)" OFF
                       SPIRV2AIR_BUILD_TRANSLATOR OFF)
include(cmake/spirv2air.cmake)
if(NOT TARGET airkit)
  add_library(airkit STATIC src/airkit/airkit.cpp)
  target_include_directories(airkit PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<INSTALL_INTERFACE:include>)
  target_compile_features(airkit PUBLIC cxx_std_17)
  target_link_libraries(airkit PUBLIC airbc mtlb)
endif()
if(NOT TARGET spirv2air::airkit)
  add_library(spirv2air::airkit ALIAS airkit)
endif()
if(SPIRV2AIR_BUILD_CLI)
  add_executable(spirv2air_cli tools/spirv2air_cli.cpp)
  target_link_libraries(spirv2air_cli PRIVATE spirv2air)
endif()

if(SPIRV2AIR_BUILD_AMBER_RUNNER)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  add_executable(amber_runner tests/amber/amber_runner.mm)
  target_link_libraries(amber_runner PRIVATE spirv2air "-framework Metal" "-framework Foundation")

  function(assemble name source environment)
    set(output "${CMAKE_CURRENT_BINARY_DIR}/test-data/${name}.spv")
    add_custom_command(OUTPUT "${output}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/test-data"
      COMMAND "${SPIRV_AS}" --target-env "${environment}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/${source}" -o "${output}"
      DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/${source}" VERBATIM)
    set(${name}_SPV "${output}" PARENT_SCOPE)
  endfunction()
